let _start_ = ago(7d);
let _end_ = now();
pageViews
| where timestamp between(_start_ .. _end_)
| summarize 
    PageLoadtime = avgif(duration, url contains "www.cineplex.com")
    by bin(timestamp, 1h)
| order by timestamp asc

    //MovieLoadtime = avgif(duration, url contains "https://www.cineplex.com/Movie/")/1000,
    TicketingLoadtime = avgif(duration, url contains "https://onlineticketing.cineplex.com/")/1000
    by bin(timestamp, 1h)
| order by timestamp asc
